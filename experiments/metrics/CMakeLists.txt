cmake_minimum_required(VERSION 3.20)
project(folly_metric_cost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(folly CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(metric_cost_bench main.cpp)
target_link_libraries(metric_cost_bench PRIVATE Folly::folly Threads::Threads)
target_compile_options(metric_cost_bench PRIVATE -Wall -Wextra)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME metric_probe_functional COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_probe.py" "$<TARGET_FILE:metric_cost_bench>")
endif()

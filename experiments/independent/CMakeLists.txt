cmake_minimum_required(VERSION 3.20)
project(independent_folly_load VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(folly CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(independent_load src/main.cpp)
target_link_libraries(independent_load PRIVATE Folly::folly Threads::Threads)
target_compile_options(independent_load PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME independent_functional COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_functional.py" "$<TARGET_FILE:independent_load>")
  add_test(NAME independent_analysis COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_analysis.py")
  add_test(NAME independent_prediction COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../stages/03-independent/test_predict.py")
  set_tests_properties(independent_functional independent_analysis PROPERTIES TIMEOUT 60)
endif()

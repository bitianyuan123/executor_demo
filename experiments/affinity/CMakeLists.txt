cmake_minimum_required(VERSION 3.20)
project(request_affinity_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(folly CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
get_filename_component(DEMO_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(request_affinity INTERFACE)
target_include_directories(request_affinity INTERFACE include ../dag/include)
target_link_libraries(request_affinity INTERFACE Folly::folly Threads::Threads)
add_executable(affinity_demo main.cpp)
target_link_libraries(affinity_demo PRIVATE request_affinity)
target_compile_options(affinity_demo PRIVATE -Wall -Wextra)
add_executable(observation_cost_bench observation_cost.cpp)
target_link_libraries(observation_cost_bench PRIVATE request_affinity)
target_compile_options(observation_cost_bench PRIVATE -Wall -Wextra)
if(BUILD_TESTING)

  foreach(name request_executor locality_workload load_policy linux_schedstat folly_active async_completion_source)

    target_link_libraries(${name}_test PRIVATE request_affinity GTest::gtest_main)
    add_test(NAME ${name}_test COMMAND ${name}_test)
    set_tests_properties(${name}_test PROPERTIES TIMEOUT 60)
  endforeach()
  add_test(NAME affinity_functional COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/functional.py" "$<TARGET_FILE:affinity_demo>")

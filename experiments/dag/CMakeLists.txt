cmake_minimum_required(VERSION 3.20)
project(folly_dag_experiment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(folly CONFIG REQUIRED)
find_package(Threads REQUIRED)
include(CTest)
option(DAG_BUILD_FOLLY_EXTENSION "Also build the opt-in stage 5 library backend" OFF)
set(FOLLY_SOURCE_DIR "" CACHE PATH "Stage 5 Folly extension source checkout (not needed for business backend)")
get_filename_component(DEMO_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

add_executable(executor_demo main.cpp)
target_include_directories(executor_demo PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(executor_demo PRIVATE Folly::folly Threads::Threads)
target_compile_options(executor_demo PRIVATE -Wall -Wextra)
# Preserve the original root build/executor_demo path for archived commands.
set_target_properties(executor_demo PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")

if(DAG_BUILD_FOLLY_EXTENSION)
  if(NOT EXISTS "${FOLLY_SOURCE_DIR}/folly/executors/experimental/AffinityExecutor.h")
    message(FATAL_ERROR "Stage 5 requires FOLLY_SOURCE_DIR with the experimental AffinityExecutor header")
  endif()
  add_executable(dag_folly main.cpp)
  target_compile_definitions(dag_folly PRIVATE EXECUTOR_DEMO_USE_FOLLY_EXTENSION=1)
  target_include_directories(dag_folly BEFORE PRIVATE "${FOLLY_SOURCE_DIR}")
  target_link_libraries(dag_folly PRIVATE Folly::folly Threads::Threads)
  target_compile_options(dag_folly PRIVATE -Wall -Wextra)
  set_target_properties(dag_folly PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()

if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(GTest REQUIRED)
  add_test(NAME demo_functional COMMAND "${Python3_EXECUTABLE}" "${DEMO_REPO_ROOT}/tests/demo_functional.py" "$<TARGET_FILE:executor_demo>")
  add_test(NAME analysis_statistics COMMAND "${Python3_EXECUTABLE}" "${DEMO_REPO_ROOT}/tests/analysis_test.py")
  add_test(NAME graph_analysis COMMAND "${Python3_EXECUTABLE}" "${DEMO_REPO_ROOT}/stages/04-dag/test_graph_analysis.py")
  set_tests_properties(demo_functional PROPERTIES TIMEOUT 120)
  add_executable(business_executor_test tests/business_executor_test.cpp)
  target_include_directories(business_executor_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_link_libraries(business_executor_test PRIVATE Folly::folly GTest::gtest_main Threads::Threads)
  add_test(NAME business_executor_test COMMAND business_executor_test)
  if(DAG_BUILD_FOLLY_EXTENSION)
    add_executable(affinity_executor_test "${FOLLY_SOURCE_DIR}/folly/executors/experimental/test/AffinityExecutorTest.cpp")
    target_include_directories(affinity_executor_test BEFORE PRIVATE "${FOLLY_SOURCE_DIR}")
    target_link_libraries(affinity_executor_test PRIVATE Folly::folly GTest::gtest_main Threads::Threads)
    add_test(NAME affinity_executor_test COMMAND affinity_executor_test)
    add_test(NAME folly_backend_functional COMMAND "${Python3_EXECUTABLE}" "${DEMO_REPO_ROOT}/tests/demo_functional.py" "$<TARGET_FILE:dag_folly>")
    add_test(NAME backend_equivalence COMMAND "${Python3_EXECUTABLE}" "${DEMO_REPO_ROOT}/stages/05-placement/compare_backends.py"
      --business "$<TARGET_FILE:executor_demo>" --library "$<TARGET_FILE:dag_folly>")
    set_tests_properties(folly_backend_functional backend_equivalence PROPERTIES TIMEOUT 120)
  endif()
endif()

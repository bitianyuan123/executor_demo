cmake_minimum_required(VERSION 3.20)
project(business_load_accounting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(folly CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(tracked_cpu_executor INTERFACE)
target_include_directories(tracked_cpu_executor INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tracked_cpu_executor INTERFACE Folly::folly Threads::Threads)
add_executable(business_accounting_example example.cpp)
target_link_libraries(business_accounting_example PRIVATE tracked_cpu_executor)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(business_accounting_test tests.cpp)
  target_link_libraries(business_accounting_test PRIVATE tracked_cpu_executor GTest::gtest_main)
  add_test(NAME business_accounting_lifecycle COMMAND business_accounting_test)
  set_tests_properties(business_accounting_lifecycle PROPERTIES TIMEOUT 30)
endif()

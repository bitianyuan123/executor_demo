cmake_minimum_required(VERSION 3.20)
project(executor_demo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(folly CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(FOLLY_SOURCE_DIR "" CACHE PATH "Folly source tree containing experimental/AffinityExecutor.h")
add_executable(executor_demo src/main.cpp)
if(FOLLY_SOURCE_DIR)
  target_include_directories(executor_demo BEFORE PRIVATE "${FOLLY_SOURCE_DIR}")
endif()
target_link_libraries(executor_demo PRIVATE Folly::folly Threads::Threads)
target_compile_options(executor_demo PRIVATE -Wall -Wextra)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME demo_functional COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/demo_functional.py" "$<TARGET_FILE:executor_demo>")
  set_tests_properties(demo_functional PROPERTIES TIMEOUT 90)
  add_test(NAME analysis_statistics COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/analysis_test.py")
  if(EXISTS "${FOLLY_SOURCE_DIR}/folly/executors/experimental/test/AffinityExecutorTest.cpp")
    find_package(GTest REQUIRED)
    add_executable(affinity_executor_test
      "${FOLLY_SOURCE_DIR}/folly/executors/experimental/test/AffinityExecutorTest.cpp")
    target_include_directories(affinity_executor_test BEFORE PRIVATE "${FOLLY_SOURCE_DIR}")
    target_link_libraries(affinity_executor_test PRIVATE Folly::folly GTest::gtest_main Threads::Threads)
    add_test(NAME affinity_executor_test COMMAND affinity_executor_test)
    set_tests_properties(affinity_executor_test PROPERTIES TIMEOUT 60)
  endif()
endif()
